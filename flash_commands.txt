# 1. Wipe Data in TWRP
# Boot to TWRP and Factory Reset. Do NOT wipe Vendor or System.

# 2. Live-boot hybris-recovery (Do NOT flash, to preserve TWRP)
# Reboot phone to bootloader/fastboot mode first
fastboot boot /opencloud/hadk/out/target/product/karatep/hybris-recovery.img

# 3. Start Python Web Server on Host PC
# (Serve from the directory containing the OS tarball and hybris-boot.img)
cd /opencloud/hadk/SailfishOScommunity-release-5.1.0.11-karatep
python3 -m http.server 8000

# 4. Connect to Sailfish Pre-Switch_Root Shell
telnet 192.168.2.15 23

# 5. Inject Commands (Copy/Paste these one by one)
tail -f /init.log &

echo "mount -t ext4 /dev/block/bootdevice/by-name/userdata /data" > /init-ctl/stdin
echo "rm -rf /data/.stowaways/sailfishos && mkdir -p /data/.stowaways/sailfishos" > /init-ctl/stdin

# (Change 192.168.2.14 to your host PC's actual USB IP if different)
echo "wget -O - http://192.168.2.14:8000/sailfishos-karatep-release-5.1.0.11.tar.bz2 | tar -xj -C /data/.stowaways/sailfishos" > /init-ctl/stdin

# Flash the kernel to boot partition
echo "wget -O /tmp/hybris-boot.img http://192.168.2.14:8000/hybris-boot.img" > /init-ctl/stdin
echo "dd if=/tmp/hybris-boot.img of=/dev/block/bootdevice/by-name/boot" > /init-ctl/stdin

echo "reboot" > /init-ctl/stdin
