# 1. Wipe Data in TWRP
# Boot to TWRP and Factory Reset. Do NOT wipe Vendor or System.

# 2. Live-boot hybris-recovery (Do NOT flash, to preserve TWRP)
# Reboot phone to bootloader/fastboot mode first
fastboot boot /opencloud/hadk/out/target/product/karatep/hybris-recovery.img

# 3. Start Python Web Server on Host PC
# (Serve from the directory containing the OS tarball and hybris-boot.img)
cd /opencloud/hadk/SailfishOScommunity-release-5.1.0.11-karatep
python3 -m http.server 8000

# 4. Connect to Sailfish Recovery Shell
telnet 192.168.2.15 2323

# 5. Run Commands directly in the terminal (Copy/Paste)
mount -t ext4 /dev/block/bootdevice/by-name/userdata /data
rm -rf /data/.stowaways/sailfishos && mkdir -p /data/.stowaways/sailfishos

# (Change 192.168.2.14 to your host PC's actual USB IP if different)
curl -f -L http://192.168.2.14:8000/sailfishos-karatep-release-5.1.0.11.tar.bz2 | tar -xj -C /data/.stowaways/sailfishos

# Flash the kernel to boot partition
curl -f -L -o /tmp/hybris-boot.img http://192.168.2.14:8000/hybris-boot.img
dd if=/tmp/hybris-boot.img of=/dev/block/bootdevice/by-name/boot

reboot
